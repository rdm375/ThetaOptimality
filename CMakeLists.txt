# The optional mixture example links GPL-3.0-or-later ThermoGPU code.
option(THETA_BUILD_GAS_EXAMPLE "Build the ThermoGPU composition example" ON)
set(THERMOGPU_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ThermoGPU" CACHE PATH "ThermoGPU source tree")
if(THETA_BUILD_GAS_EXAMPLE)
    add_executable(peng_robinson_inverse examples/peng_robinson_inverse.cpp
        ${THERMOGPU_ROOT}/src/cpu/pr_scalar.cpp)
    target_include_directories(peng_robinson_inverse PRIVATE ${THERMOGPU_ROOT}/include)
    target_link_libraries(peng_robinson_inverse PRIVATE theta_opt)
    if(BUILD_TESTING)
        add_test(NAME peng_robinson_composition_qualification COMMAND peng_robinson_inverse ${CMAKE_CURRENT_BINARY_DIR}/gas_results)
    endif()
    add_executable(time_varying_gas examples/time_varying_gas.cpp
        ${THERMOGPU_ROOT}/src/cpu/pr_scalar.cpp)
    target_include_directories(time_varying_gas PRIVATE ${THERMOGPU_ROOT}/include)
    target_link_libraries(time_varying_gas PRIVATE theta_opt)
    if(BUILD_TESTING)
        add_test(NAME time_varying_gas_composition_reconstruction
            COMMAND time_varying_gas ${CMAKE_CURRENT_BINARY_DIR}/time_varying_gas_results)
    endif()
endif()
if(BUILD_TESTING)
    add_executable(theta_validation tests/validation.cpp)
    target_link_libraries(theta_validation PRIVATE theta_opt)
    add_test(NAME discrete_optimality_validation COMMAND theta_validation)
    add_test(NAME electrothermal_global_ambiguity COMMAND electrothermal ${CMAKE_CURRENT_BINARY_DIR}/electrothermal_results)
    add_test(NAME insulated_thermal_unobservable_mode COMMAND insulated_thermal ${CMAKE_CURRENT_BINARY_DIR}/insulated_thermal_results)

    add_test(NAME oscillator_poor_timing_nonidentifiability COMMAND oscillator_poor_timing ${CMAKE_CURRENT_BINARY_DIR}/oscillator_poor_timing_results)
    add_test(NAME oscillator_proper_timing_recovery COMMAND oscillator_poor_timing ${CMAKE_CURRENT_BINARY_DIR}/oscillator_proper_timing_results proper)
    add_test(NAME oscillator_weak_information_sweep COMMAND oscillator_information_sweep ${CMAKE_CURRENT_BINARY_DIR}/oscillator_information_sweep_results)
